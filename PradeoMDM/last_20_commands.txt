// upload the VendorCertificateRequest.csr and then download vendor certificate on apple dev and upload it to the container
docker cp test/mdm-2.cer pradeomdm-micromdm-1:/app/mdm-certificates/

// create a push CSR (PushCertificateRequest.key and PushCertificatePrivateKey.key)
docker exec -it pradeomdm-micromdm-1 ./mdmctl mdmcert push -password=vlkasvkja1rd -country=FR -email=leo.brunet@pradeo.com
docker exec -it pradeomdm-micromdm-1 ls -lt  mdm-certificates

// sign the push certificate request with the vendor certificate which generate PushCertificateRequest.plist
docker exec -it pradeomdm-micromdm-1 ./mdmctl mdmcert vendor -sign -cert=./mdm-certificates/mdm-2.cer -password=vlkasvkja1rd
docker exec -it pradeomdm-micromdm-1 ls -lt  mdm-certificates

// get the PushCertificateRequest.plist from the container
docker cp pradeomdm-micromdm-1:/app/mdm-certificates/PushCertificateRequest.plist ./test/

// upload the PushCertificateRequest.plist file to get the APNS certificate and download MDM_Pradeo_Security_Systems_Certificate.pem and upload it to the container
docker cp test/MDM_Cert.pem pradeomdm-micromdm-1:/app/mdm-certificates/

// upload the certificate to MicroMDM
docker exec -it pradeomdm-micromdm-1 ./mdmctl mdmcert upload -cert ./mdm-certificates/MDM_Cert.pem  -private-key mdm-certificates/PushCertificatePrivateKey.key -password=vlkasvkja1rd
